/**
 * 중복 생년월일 그룹 매칭 검증
 */
function validateDuplicateBirthdateMatching() {
  const testCases = [
    {
      group: '570508',
      employees: [
        { empNo: 176, name: '이순애', residentId: '570508-2156427' },
        { empNo: 212, name: '김연숙', residentId: '570508-2156432' }
      ]
    },
    {
      group: '591225',
      employees: [
        { empNo: 6, name: '변광연', residentId: '591225-1661113' },
        { empNo: 244, name: '이상분', residentId: '591225-2792911' }
      ]
    },
    {
      group: '601205',
      employees: [
        { empNo: 209, name: '이순화', residentId: '601205-2656919' },
        { empNo: 238, name: '최병말', residentId: '601205-2483213' }
      ]
    }
  ];

  const month = '2026-01-31';
  const results = [];

  testCases.forEach(testCase => {
    const groupResults = [];

    testCase.employees.forEach(emp => {
      const insuranceData = matchInsuranceData(
        getInsuranceSheets(),
        month,
        emp.residentId,
        emp.name  // 이름 전달
      );

      groupResults.push({
        empNo: emp.empNo,
        name: emp.name,
        healthInsurance: insuranceData.healthInsurance,
        longTermCare: insuranceData.longTermCare,
        employeeInsurance: insuranceData.employeeInsurance,
        employerInsurance: insuranceData.employerInsurance
      });
    });

    // 두 근로자의 보험료가 서로 다른지 확인
    if (groupResults.length === 2) {
      const isDifferent =
        groupResults[0].healthInsurance !== groupResults[1].healthInsurance ||
        groupResults[0].longTermCare !== groupResults[1].longTermCare;

      results.push({
        group: testCase.group,
        passed: isDifferent,
        data: groupResults
      });
    }
  });

  return results;
}
