## ✅ 주요 변경 사항

### 📊 컬럼 구조 변경

## 🆕 신규 추가 컬럼 (3개)

### 1. AF열 - 급여 지급 은행
```
- 열명: 급여 지급 은행
- 데이터 타입: 텍스트
- 필수 여부: 필수
- 설명: 급여 지급 은행명
- 예시: "국민은행", "신한은행", "우리은행"
- 비고: 정확한 은행명 표기 필요 (약칭 사용 주의)
```

### 2. AG열 - 계좌 번호
```
- 열명: 계좌 번호
- 데이터 타입: 텍스트
- 필수 여부: 필수
- 설명: 급여 지급 계좌번호
- 예시: "123-456-789012"
- 비고: ⚠️ 민감정보 - 암호화 저장 강력 권장
```

### 3. AH열 - 예금주
```
- 열명: 예금주
- 데이터 타입: 텍스트
- 필수 여부: 필수
- 설명: 계좌 예금주명
- 예시: "홍길동"
- 비고: 근무자명(B열)과 일치 여부 검증 필요
```

```javascript
function validate예금주일치(근무자명, 예금주) {
  // 공백 제거 후 비교
  const 정규화근무자명 = 근무자명.replace(/\s/g, '');
  const 정규화예금주 = 예금주.replace(/\s/g, '');

  if (정규화근무자명 !== 정규화예금주) {
    console.warn('예금주 불일치:', {
      근무자명: 근무자명,
      예금주: 예금주
    });
    return false;
  }

  return true;
}
```

### 계좌번호 형식 검증
```javascript
function validate계좌번호형식(계좌번호) {
  // 숫자와 하이픈만 허용
  const pattern = /^[0-9\-]+$/;

  if (!pattern.test(계좌번호)) {
    console.warn('잘못된 계좌번호 형식:', 계좌번호);
    return false;
  }

  // 최소 길이 체크 (10자리 이상)
  const digitsOnly = 계좌번호.replace(/\-/g, '');
  if (digitsOnly.length < 10) {
    console.warn('계좌번호가 너무 짧음:', 계좌번호);
    return false;
  }

  return true;
}
```

```javascript
function 급여이체처리(직원목록) {

    };

    // 검증
    if (!validate예금주일치(직원[급여데이터_컬럼매핑.근무자명], 이체정보.예금주)) {

    // 이체 처리
```

### 2. 계좌정보 마스킹 표시

```javascript
function mask계좌번호(계좌번호) {
  // 앞 3자리와 뒤 4자리만 표시
  if (계좌번호.length < 7) return '***';

  const 앞부분 = 계좌번호.substring(0, 3);
  const 뒷부분 = 계좌번호.substring(계좌번호.length - 4);

  return `${앞부분}****${뒷부분}`;
}

// 사용 예시
console.log(mask계좌번호('123-456-789012')); // "123****9012"
```
