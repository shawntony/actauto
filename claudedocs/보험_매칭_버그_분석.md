### 방안 3: 중복 매칭 감지 및 경고

#### 장점
- 기존 로직 유지하면서 문제 감지
- 사용자에게 명확한 오류 메시지 제공

#### 단점
- 근본적 해결책이 아님
- 여전히 잘못된 매칭 발생 가능

#### 구현
```javascript
function findInsuranceValue(sheet, month, idKey, schema) {
  const matches = [];

  for (const row of data) {
    const monthValue = String(row[0]).trim();
    const rowIdKey = extractResidentIdKey(String(row[idColIndex]));

    if (monthValue === month && rowIdKey === idKey) {
      matches.push({
        name: row[nameColIndex],
        value: row[dataColIndex]
      });
    }
  }

  if (matches.length === 0) {
    return 0;
  } else if (matches.length === 1) {
    return matches[0].value;
  } else {
    // ⚠️ 중복 매칭 경고
    Logger.log('[경고] 동일 생년월일 ' + idKey + '로 ' + matches.length + '명 매칭됨: ' +
               matches.map(m => m.name).join(', '));
    throw new Error('중복 매칭: ' + idKey + ' - ' +
                    matches.map(m => m.name).join(', '));
  }
}
```

## 🎯 권장 해결책

**방안 1 (주민번호 전체 사용) + 방안 3 (중복 감지)**를 조합:

```javascript
/**
 * 주민번호에서 매칭 키 추출
 * @param {string} residentId - 주민등록번호
 * @param {boolean} fullId - true: 전체 번호, false: 앞 6자리 (기본값: true)
 * @returns {string} 매칭 키
 */
function extractResidentIdKey(residentId, fullId = true) {
  if (!residentId) return '';
  const cleaned = String(residentId).replace(/[-\s]/g, '');

  if (fullId) {
    return cleaned;  // 전체 주민번호 (13자리)
  } else {
    return cleaned.substring(0, 6);  // 앞 6자리만 (하위 호환성)
  }
}

/**
 * 보험 시트에서 단일 값 찾기 (중복 감지 포함)
 */
function findInsuranceValue(sheet, month, residentId, schema) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return 0;

  const idColIndex = columnLetterToIndex(schema.MATCHING_COLUMNS.RESIDENT_ID);
  const dataCol = Object.values(schema.DATA_COLUMNS)[0];
  const dataColIndex = columnLetterToIndex(dataCol);
  const maxCol = Math.max(idColIndex, dataColIndex) + 1;

  const data = sheet.getRange(2, 1, lastRow - 1, maxCol).getValues();
  const matches = [];

  // 전체 주민번호로 매칭
  const fullIdKey = extractResidentIdKey(residentId, true);

  for (const row of data) {
    const monthValue = String(row[0]).trim();
    const rowIdKey = extractResidentIdKey(String(row[idColIndex]), true);

    if (monthValue === month && rowIdKey === fullIdKey) {
      matches.push({
        residentId: row[idColIndex],
        value: row[dataColIndex] || 0
      });
    }
  }

  if (matches.length === 0) {
    Logger.log('[경고] 매칭 실패: ' + month + ', ' + residentId);
    return 0;
  } else if (matches.length === 1) {
    return matches[0].value;
  } else {
    // 중복 매칭 오류
    throw new Error('중복 매칭 오류: ' + residentId + ' (' + matches.length + '건)');
  }
}
```
