### 컬럼 문자 변환 함수

```javascript
// 인덱스를 컬럼 문자로 변환
function indexToColumn(index) {
  let column = '';
  let temp = index;
  while (temp >= 0) {
    column = String.fromCharCode((temp % 26) + 65) + column;
    temp = Math.floor(temp / 26) - 1;
  }
  return column;
}

// 컬럼 문자를 인덱스로 변환
function columnToIndex(column) {
  let index = 0;
  for (let i = 0; i < column.length; i++) {
    index = index * 26 + (column.charCodeAt(i) - 64);
  }
  return index - 1;
}

// 사용 예시
console.log(indexToColumn(0));   // "A"
console.log(indexToColumn(30));  // "AE"
console.log(indexToColumn(33));  // "AH"
console.log(columnToIndex('A')); // 0
console.log(columnToIndex('AE')); // 30
console.log(columnToIndex('AH')); // 33
```

## ⚠️ 주의사항

### 데이터 무결성

1. **주민등록번호 (C열)**
   - 민감정보로 반드시 보안 처리 필요
   - 암호화 저장 권장
   - 접근 권한 엄격히 관리

2. **계약시작일 (F열)**
   - Date 객체 또는 "YYYY.MM.DD" 형식 문자열
   - 4대보험신고 시 "YYYYMMDD" 형식으로 변환 필요

3. **월보수 (S열)**
   - 4대보험 기준 금액으로 정확성 매우 중요
   - 실제 지급액이 아닌 보험료 산정 기준액

4. **1주근로시간 (AC열)**
   - 법정 근로시간 범위 검증 필요 (주 40시간 기준)
   - 4대보험신고 시 사용되므로 정확해야 함

5. **적용월 (T열)**
   - 급여 적용 기준 월 (예: "2026-01")
   - ⚠️ 주의: 이전에 "직종코드"로 잘못 기록된 적 있음

6. **메모 (AD열)**
   - 급여 관련 특이사항 기록용
   - 계산에 영향 없는 참고용 필드

7. **통상임금 (AE열)**
   - 법정 통상임금 계산 결과
   - 연차수당, 연장수당 계산 기준

8. **급여 지급 은행 (AF열)**
   - 급여 이체 시 사용되는 은행명
   - 정확한 은행명 표기 필요 (약칭 사용 주의)

9. **계좌 번호 (AG열)**
   - 민감정보로 반드시 보안 처리 필요
   - 암호화 저장 강력 권장
   - 접근 권한 엄격히 관리

10. **예금주 (AH열)**
    - 근무자명(B열)과 일치 여부 검증 필요
    - 불일치 시 급여 이체 오류 발생 가능

```javascript
// Google Sheets API 사용
const spreadsheetId = 'YOUR_SPREADSHEET_ID';
```

```javascript
sheets.spreadsheets.values.update({
  spreadsheetId: spreadsheetId,
  range: range,
  valueInputOption: 'USER_ENTERED',
  resource: { values: values }

```

### 3. 급여 계산 검증

```javascript
// 실_월지급액 계산 검증
function validate급여계산(rowData) {
  const 실지급합계 =
    Number(rowData[급여데이터_컬럼매핑.실_기본급]) +
    Number(rowData[급여데이터_컬럼매핑.실_연장수당]) +
    Number(rowData[급여데이터_컬럼매핑.실_연차수당]) +
    Number(rowData[급여데이터_컬럼매핑.실_직책수당]) +
    Number(rowData[급여데이터_컬럼매핑.실_공휴일특근수당]);

  const 기록된실지급액 = Number(rowData[급여데이터_컬럼매핑.실_월지급액]);

  if (Math.abs(실지급합계 - 기록된실지급액) > 1) {
    console.warn('급여 계산 오류 발견:', {
      계산값: 실지급합계,
      기록값: 기록된실지급액,
      차이: 실지급합계 - 기록된실지급액
    });
    return false;
  }

  return true;
}
```
